    <script>
        // Initialize Telegram WebApp SDK
        const tg = window.Telegram ? window.Telegram.WebApp : null;

        if (tg) {
            tg.ready();
            tg.expand(); // Make Web App full height in Telegram
        }

        // Product List Data
        const products = [
            { id: 1, name: "សេះមាសលើកជើងពីរ", price: 4.08, shipping: 1.86, icon: "🐎", checked: true },
            { id: 2, name: "ស្លាបហង្សមាស ខ្សាច់ហូរ", price: 4.15, shipping: 1.56, icon: "🪶", checked: true },
            { id: 3, name: "ឥន្ទ្រី mini", price: 4.82, shipping: 0.98, icon: "🦅", checked: true },
            { id: 4, name: "សំពៅមាស", price: 3.52, shipping: 1.50, icon: "⛵", checked: true }
        ];

        // Render Product List
        function renderProducts() {
            const container = document.getElementById('product-list');
            container.innerHTML = '';

            products.forEach(product => {
                const totalItemPrice = (product.price + product.shipping).toFixed(2);
                
                const card = document.createElement('div');
                card.className = `tg-card rounded-xl p-3.5 border transition-all duration-200 flex items-center gap-3 cursor-pointer select-none ${
                    product.checked ? 'border-blue-500/50 shadow-sm' : 'opacity-70 border-gray-200'
                }`;
                card.onclick = (e) => {
                    // Prevent double toggling if clicking directly on checkbox
                    if (e.target.tagName !== 'INPUT') {
                        toggleProduct(product.id);
                    }
                };

                card.innerHTML = `
                    <div>
                        <input type="checkbox" class="custom-checkbox" ${product.checked ? 'checked' : ''} onchange="toggleProduct(${product.id})">
                    </div>
                    <div class="text-2xl">${product.icon}</div>
                    <div class="flex-1 min-w-0">
                        <h4 class="font-semibold text-sm truncate">${product.name}</h4>
                        <div class="text-xs tg-hint mt-0.5 flex gap-2">
                            <span>តម្លៃ៖ $${product.price.toFixed(2)}</span>
                            <span>•</span>
                            <span>ដឹក៖ $${product.shipping.toFixed(2)}</span>
                        </div>
                    </div>
                    <div class="text-right">
                        <div class="font-bold text-sm text-blue-600 dark:text-blue-400">$${totalItemPrice}</div>
                        <div class="text-[10px] tg-hint">សរុប</div>
                    </div>
                `;
                container.appendChild(card);
            });

            calculateTotal();
        }

        // Toggle Single Checkbox
        function toggleProduct(id) {
            const product = products.find(p => p.id === id);
            if (product) {
                product.checked = !product.checked;
                renderProducts();
            }
        }

        // Select All / Deselect All
        function toggleSelectAll(status) {
            products.forEach(p => p.checked = status);
            renderProducts();
        }

        // Calculate Totals
        function calculateTotal() {
            let selectedCount = 0;
            let totalItemsPrice = 0;
            let totalShippingPrice = 0;

            products.forEach(p => {
                if (p.checked) {
                    selectedCount++;
                    totalItemsPrice += p.price;
                    totalShippingPrice += p.shipping;
                }
            });

            const grandTotal = totalItemsPrice + totalShippingPrice;

            document.getElementById('selected-count').innerText = `${selectedCount} មុខ`;
            document.getElementById('total-items-price').innerText = `$${totalItemsPrice.toFixed(2)}`;
            document.getElementById('total-shipping-price').innerText = `$${totalShippingPrice.toFixed(2)}`;
            document.getElementById('grand-total').innerText = `$${grandTotal.toFixed(2)}`;
        }

        // Show Notification Toast
        function showToast(message) {
            const toast = document.getElementById('toast');
            toast.innerText = message;
            toast.classList.remove('opacity-0');
            toast.classList.add('opacity-100');
            setTimeout(() => {
                toast.classList.remove('opacity-100');
                toast.classList.add('opacity-0');
            }, 2500);
        }

        // Send Order / Copy Order
        function sendOrderToTelegram() {
            const selectedItems = products.filter(p => p.checked);

            if (selectedItems.length === 0) {
                if (tg && tg.showAlert) {
                    tg.showAlert("សូមជ្រើសរើសទំនិញយ៉ាងហោចណាស់មួយ!");
                } else {
                    alert("សូមជ្រើសរើសទំនិញយ៉ាងហោចណាស់មួយ!");
                }
                return;
            }

            let orderText = "🛒 **ការបញ្ជាទិញថ្មីពី Web App**\n---------------------------\n";
            let totalItems = 0;
            let totalShip = 0;

            selectedItems.forEach((item, index) => {
                const subtotal = item.price + item.shipping;
                totalItems += item.price;
                totalShip += item.shipping;
                orderText += `${index + 1}. ${item.name}\n   - តម្លៃ: $${item.price.toFixed(2)} | ដឹក: $${item.shipping.toFixed(2)} => **$${subtotal.toFixed(2)}**\n`;
            });

            const grandTotal = totalItems + totalShip;
            orderText += "---------------------------\n";
            orderText += `📦 សរុបទំនិញ: $${totalItems.toFixed(2)}\n`;
            orderText += `🚚 សរុបថ្លៃដឹក: $${totalShip.toFixed(2)}\n`;
            orderText += `💰 **តម្លៃសរុបរួម: $${grandTotal.toFixed(2)}**`;

            // If inside Telegram WebApp, send data back to bot
            if (tg && tg.sendData) {
                try {
                    tg.sendData(JSON.stringify({
                        items: selectedItems,
                        totalItemsPrice: totalItems.toFixed(2),
                        totalShippingPrice: totalShip.toFixed(2),
                        grandTotal: grandTotal.toFixed(2)
                    }));
                    tg.close();
                } catch (e) {
                    copyToClipboard(orderText);
                }
            } else {
                // Fallback for normal browsers: Copy to clipboard
                copyToClipboard(orderText);
            }
        }

        function copyToClipboard(text) {
            const textarea = document.createElement('textarea');
            textarea.value = text;
            document.body.appendChild(textarea);
            textarea.select();
            document.execCommand('copy');
            document.body.removeChild(textarea);
            showToast("បានចម្លងអត្ថបទការកម្មង់! អាច Paste ផ្ញើក្នុង Chat បាន");
        }

        // Initial Render
        renderProducts();
    </script>
